<script>
    // ============================================================
    //  ДНЕВНИК ДОЛГОВ — Vanilla JS
    // ============================================================

    // ---------- Ключи localStorage ----------
    const LS_KEYS = {
        USERS: 'debt_users',
        SUBJECTS: 'debt_subjects',
        DATA: 'debt_data',
        SESSION: 'debt_session'
    };

    // ---------- Демо-данные (при первом запуске) ----------
    const DEFAULT_USERS = [
        { id: 'u1', login: 'admin', password: 'admin123', role: 'admin', name: 'Администратор' },
        { id: 'u2', login: 'ivanov', password: '1234', role: 'student', name: 'Иванов Иван Иванович' },
        { id: 'u3', login: 'petrov', password: '1234', role: 'student', name: 'Петров Пётр Петрович' },
        { id: 'u4', login: 'sidorova', password: '1234', role: 'student', name: 'Сидорова Анна Сергеевна' },
        { id: 'u5', login: 'kuznetsov', password: '1234', role: 'student', name: 'Кузнецов Дмитрий Олегович' }
    ];

    const DEFAULT_SUBJECTS = [
        { id: 's1', name: 'Математика' },
        { id: 's2', name: 'Физика' },
        { id: 's3', name: 'Информатика' },
        { id: 's4', name: 'История' }
    ];

    // Ключи: id ученика -> id предмета -> текст долга
    const DEFAULT_DATA = {
        'u2': { 's1': '2 курс 2 семестр', 's3': '1 курс 1 семестр' },
        'u3': { 's2': '3 курс 1 семестр', 's4': '2 курс 2 семестр' },
        'u4': { 's1': '1 курс 1 семестр' },
        'u5': { 's2': '2 курс 2 семестр', 's3': '3 курс 1 семестр', 's4': '1 курс 2 семестр' }
    };

    // ---------- Хранилище ----------
    let users = [];
    let subjects = [];
    let debtData = {};       // { studentId: { subjectId: text } }
    let currentUser = null;  // объект пользователя или null

    // ---------- Вспомогательные функции ----------
    function loadFromLS(key, fallback) {
        try {
            const raw = localStorage.getItem(key);
            if (raw) return JSON.parse(raw);
        } catch (e) {
            console.warn('Ошибка чтения localStorage', e);
        }
        return fallback;
    }

    function saveToLS(key, value) {
        try {
            localStorage.setItem(key, JSON.stringify(value));
        } catch (e) {
            console.warn('Ошибка записи в localStorage', e);
        }
    }

    // Генерация id
    function genId(prefix = 'id') {
        return prefix + '_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8);
    }

    // Toast-уведомление
    function showToast(message) {
        const container = document.getElementById('toastContainer');
        const toast = document.createElement('div');
        toast.className = 'toast';
        toast.textContent = message;
        container.appendChild(toast);
        setTimeout(() => {
            toast.style.opacity = '0';
            toast.style.transition = 'opacity 0.2s';
            setTimeout(() => toast.remove(), 250);
        }, 2600);
    }

    // ---------- Инициализация данных ----------
    function initStorage() {
        // Пользователи
        if (!localStorage.getItem(LS_KEYS.USERS)) {
            saveToLS(LS_KEYS.USERS, DEFAULT_USERS);
        }
        users = loadFromLS(LS_KEYS.USERS, DEFAULT_USERS);

        // Предметы
        if (!localStorage.getItem(LS_KEYS.SUBJECTS)) {
            saveToLS(LS_KEYS.SUBJECTS, DEFAULT_SUBJECTS);
        }
        subjects = loadFromLS(LS_KEYS.SUBJECTS, DEFAULT_SUBJECTS);

        // Данные таблицы
        if (!localStorage.getItem(LS_KEYS.DATA)) {
            saveToLS(LS_KEYS.DATA, DEFAULT_DATA);
        }
        debtData = loadFromLS(LS_KEYS.DATA, DEFAULT_DATA);

        // Сессия
        const sessionRaw = localStorage.getItem(LS_KEYS.SESSION);
        if (sessionRaw) {
            try {
                const session = JSON.parse(sessionRaw);
                if (session && session.userId) {
                    const user = users.find(u => u.id === session.userId);
                    if (user) currentUser = user;
                }
            } catch (e) { /* ignore */ }
        }
    }

    // ---------- Сохранение отдельных частей ----------
    function saveUsers() { saveToLS(LS_KEYS.USERS, users); }
    function saveSubjects() { saveToLS(LS_KEYS.SUBJECTS, subjects); }
    function saveData() { saveToLS(LS_KEYS.DATA, debtData); }

    // ---------- Работа с сессией ----------
    function saveSession(user) {
        if (user) {
            saveToLS(LS_KEYS.SESSION, { userId: user.id });
        } else {
            localStorage.removeItem(LS_KEYS.SESSION);
        }
    }

    // ---------- Показать/скрыть экраны ----------
    function showLoginScreen() {
        document.getElementById('loginScreen').classList.remove('hidden');
        document.getElementById('mainScreen').classList.add('hidden');
        // Сброс полей
        document.getElementById('loginInput').value = '';
        document.getElementById('passwordInput').value = '';
        document.getElementById('loginError').textContent = '';
    }

    function showMainScreen() {
        document.getElementById('loginScreen').classList.add('hidden');
        document.getElementById('mainScreen').classList.remove('hidden');
        // Обновить шапку
        document.getElementById('headerUserName').textContent = currentUser.name || currentUser.login;
        document.getElementById('headerUserRole').textContent = currentUser.role === 'admin' ? 'Администратор' : 'Ученик';
        // Панель инструментов — только для админа
        const toolbar = document.getElementById('adminToolbar');
        if (currentUser.role === 'admin') {
            toolbar.classList.remove('hidden');
        } else {
            toolbar.classList.add('hidden');
        }
        renderTable();
    }

    // ---------- Выход ----------
    function logout() {
        currentUser = null;
        saveSession(null);
        showLoginScreen();
        showToast('Вы вышли из системы');
    }

    // ============================================================
    //  РЕНДЕРИНГ ТАБЛИЦЫ
    // ============================================================

    function renderTable() {
        if (!currentUser) return;

        const thead = document.getElementById('tableHead');
        const tbody = document.getElementById('tableBody');
        const isAdmin = currentUser.role === 'admin';

        // --- Заголовок ---
        thead.innerHTML = '';

        // Первый столбец "Ученик"
        const headRow = document.createElement('tr');
        const thStudent = document.createElement('th');
        thStudent.textContent = 'Ученик';
        thStudent.style.minWidth = '180px';
        headRow.appendChild(thStudent);

        // Столбцы предметов
        subjects.forEach(subject => {
            const th = document.createElement('th');
            th.style.minWidth = '150px';

            const container = document.createElement('div');
            container.style.display = 'flex';
            container.style.alignItems = 'center';
            container.style.justifyContent = 'space-between';
            container.style.gap = '0.5rem';

            const nameSpan = document.createElement('span');
            nameSpan.textContent = subject.name;
            container.appendChild(nameSpan);

            if (isAdmin) {
                const delBtn = document.createElement('button');
                delBtn.className = 'icon-btn';
                delBtn.innerHTML = '✕';
                delBtn.title = 'Удалить предмет';
                delBtn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    askDeleteSubject(subject.id, subject.name);
                });
                container.appendChild(delBtn);
            }

            th.appendChild(container);
            headRow.appendChild(th);
        });

        // Столбец действий (только для админа) — для удаления строк
        if (isAdmin) {
            const thActions = document.createElement('th');
            thActions.textContent = '⚙️';
            thActions.style.minWidth = '60px';
            thActions.style.textAlign = 'center';
            headRow.appendChild(thActions);
        }

        thead.appendChild(headRow);

        // --- Тело таблицы ---
        tbody.innerHTML = '';

        // Фильтрация: ученик видит только свою строку
        let visibleUsers = users.filter(u => u.role === 'student');
        if (!isAdmin) {
            // Только текущий ученик
            visibleUsers = visibleUsers.filter(u => u.id === currentUser.id);
        }

        if (visibleUsers.length === 0) {
            // Ничего не показываем, но можно добавить сообщение
            const emptyRow = document.createElement('tr');
            const td = document.createElement('td');
            td.colSpan = subjects.length + 1 + (isAdmin ? 1 : 0);
            td.textContent = 'Нет учеников для отображения';
            td.style.textAlign = 'center';
            td.style.color = '#94a3b8';
            emptyRow.appendChild(td);
            tbody.appendChild(emptyRow);
        }

        visibleUsers.forEach(user => {
            const row = document.createElement('tr');

            // Ячейка с ФИО
            const nameTd = document.createElement('td');
            nameTd.textContent = user.name;
            nameTd.style.fontWeight = '500';
            row.appendChild(nameTd);

            // Ячейки по предметам
            subjects.forEach(subject => {
                const td = document.createElement('td');
                const value = (debtData[user.id] && debtData[user.id][subject.id]) || '';

                if (isAdmin) {
                    // Админ может редактировать ячейку по двойному клику
                    td.style.cursor = 'pointer';
                    td.title = 'Двойной клик для редактирования';

                    const span = document.createElement('span');
                    span.textContent = value || '—';
                    td.appendChild(span);

                    // Классы для подсветки
                    if (value) {
                        td.classList.add('debt-cell');
                    } else {
                        td.classList.add('empty-cell');
                    }

                    // Двойной клик — inline-редактирование
                    td.addEventListener('dblclick', () => startInlineEdit(td, user.id, subject.id, value));

                    row.appendChild(td);
                } else {
                    // Ученик — только чтение
                    td.textContent = value || '—';
                    if (value) {
                        td.classList.add('debt-cell');
                    } else {
                        td.classList.add('empty-cell');
                    }
                    row.appendChild(td);
                }
            });

            // Столбец действий (удаление строки) — только админ
            if (isAdmin) {
                const actionsTd = document.createElement('td');
                actionsTd.style.textAlign = 'center';

                const delBtn = document.createElement('button');
                delBtn.className = 'icon-btn';
                delBtn.innerHTML = '🗑️';
                delBtn.title = 'Удалить ученика';
                delBtn.addEventListener('click', () => askDeleteStudent(user.id, user.name));
                actionsTd.appendChild(delBtn);

                row.appendChild(actionsTd);
            }

            tbody.appendChild(row);
        });
    }

    // ---------- Inline-редактирование ячейки ----------
    function startInlineEdit(td, studentId, subjectId, currentValue) {
        // Не даём открыть два редактора одновременно
        if (td.querySelector('input')) return;

        const originalHTML = td.innerHTML;
        td.innerHTML = '';

        const input = document.createElement('input');
        input.type = 'text';
        input.className = 'edit-input';
        input.value = currentValue;
        input.placeholder = 'Нет долга';

        const saveBtn = document.createElement('button');
        saveBtn.className = 'icon-btn';
        saveBtn.innerHTML = '✔️';
        saveBtn.title = 'Сохранить';
        saveBtn.style.marginLeft = '0.25rem';

        const cancelBtn = document.createElement('button');
        cancelBtn.className = 'icon-btn';
        cancelBtn.innerHTML = '✖️';
        cancelBtn.title = 'Отмена';
        cancelBtn.style.marginLeft = '0.1rem';

        const wrapper = document.createElement('div');
        wrapper.style.display = 'flex';
        wrapper.style.alignItems = 'center';
        wrapper.appendChild(input);
        wrapper.appendChild(saveBtn);
        wrapper.appendChild(cancelBtn);
        td.appendChild(wrapper);

        input.focus();
        input.select();

        function save() {
            const newValue = input.value.trim();
            if (!debtData[studentId]) debtData[studentId] = {};
            if (newValue) {
                debtData[studentId][subjectId] = newValue;
            } else {
                delete debtData[studentId][subjectId];
                // Если у ученика больше нет долгов — удаляем пустой объект
                if (Object.keys(debtData[studentId]).length === 0) {
                    delete debtData[studentId];
                }
            }
            saveData();
            renderTable();
            showToast('Сохранено');
        }

        function cancel() {
            td.innerHTML = originalHTML;
            // Восстановить классы (на случай, если они были сброшены)
            // Проще перерисовать всю таблицу, но можно вручную:
            // Лучше перерисовать — надёжнее.
            renderTable();
        }

        saveBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            save();
        });

        cancelBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            cancel();
        });

        input.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') {
                e.preventDefault();
                save();
            } else if (e.key === 'Escape') {
                e.preventDefault();
                cancel();
            }
        });

        // Клик вне — отмена
        const clickOutside = (e) => {
            if (!td.contains(e.target)) {
                cancel();
                document.removeEventListener('click', clickOutside);
            }
        };
        setTimeout(() => document.addEventListener('click', clickOutside), 0);
    }

    // ============================================================
    //  CRUD: УЧЕНИКИ
    // ============================================================

    function addStudent(name, login, password) {
        // Валидация
        if (!name.trim() || !login.trim() || !password.trim()) {
            return { ok: false, error: 'Заполните все поля' };
        }

        // Уникальность логина
        if (users.some(u => u.login.toLowerCase() === login.trim().toLowerCase())) {
            return { ok: false, error: 'Пользователь с таким логином уже существует' };
        }

        const newUser = {
            id: genId('u'),
            login: login.trim(),
            password: password.trim(),
            role: 'student',
            name: name.trim()
        };

        users.push(newUser);
        saveUsers();

        // Инициализируем пустую запись в данных (чтобы ученик отображался)
        if (!debtData[newUser.id]) {
            debtData[newUser.id] = {};
            saveData();
        }

        showToast('Ученик добавлен');
        renderTable();
        return { ok: true };
    }

    function deleteStudent(studentId) {
        // Удаляем пользователя
        users = users.filter(u => u.id !== studentId);
        saveUsers();

        // Удаляем его данные из таблицы
        delete debtData[studentId];
        saveData();

        // Если удалили себя (админ не может удалить себя — проверим)
        if (currentUser && currentUser.id === studentId) {
            logout();
        }

        showToast('Ученик удалён');
        renderTable();
    }

    //  CRUD: ПРЕДМЕТЫ
    // ============================================================

    function addSubject(name) {
        if (!name.trim()) {
            return { ok: false, error: 'Введите название предмета' };
        }

        // Проверка на дубликат (не строго, но желательно)
        if (subjects.some(s => s.name.toLowerCase() === name.trim().toLowerCase())) {
            return { ok: false, error: 'Такой предмет уже существует' };
        }

        const newSubject = {
            id: genId('s'),
            name: name.trim()
        };

        subjects.push(newSubject);
        saveSubjects();

        showToast('Столбец добавлен');
        renderTable();
        return { ok: true };
    }

    function deleteSubject(subjectId, subjectName) {
        // Удаляем предмет
        subjects = subjects.filter(s => s.id !== subjectId);
        saveSubjects();

        // Удаляем все ячейки с этим предметом
        for (const studentId in debtData) {
            if (debtData[studentId] && debtData[studentId][subjectId]) {
                delete debtData[studentId][subjectId];
                // Если у ученика не осталось долгов — удаляем пустой объект
                if (Object.keys(debtData[studentId]).length === 0) {
                    delete debtData[studentId];
                }
            }
        }
        saveData();

        showToast(`Предмет «${subjectName}» удалён`);
        renderTable();
    }

    const cancelConfirmBtn = document.getElementById('cancelConfirmBtn');

    let pendingDeleteAction = null;

    function askDeleteStudent(studentId, studentName) {
        confirmTitle.textContent = 'Удалить ученика?';
        confirmMessage.textContent = `Вы уверены, что хотите удалить «${studentName}»? Все данные о его долгах будут потеряны.`;

        confirmModal.classList.remove('hidden');
    }

    function askDeleteSubject(subjectId, subjectName) {
        confirmTitle.textContent = 'Удалить предмет?';
        confirmMessage.textContent = `Вы уверены, что хотите удалить столбец «${subjectName}»? Все долги по этому предмету будут удалены.`;
        pendingDeleteAction = () => deleteSubject(subjectId, subjectName);
        confirmModal.classList.remove('hidden');
    }

        if (pendingDeleteAction) {
            pendingDeleteAction();
            pendingDeleteAction = null;
        }
        confirmModal.classList.add('hidden');
    });

    cancelConfirmBtn.addEventListener('click', () => {
        pendingDeleteAction = null;
        confirmModal.classList.add('hidden');
    });

    [studentModal, subjectModal, confirmModal].forEach(modal => {
        modal.addEventListener('click', (e) => {
            if (e.target === modal) {
                modal.classList.add('hidden');
                pendingDeleteAction = null;
            }
        });

            loginError.textContent = 'Неверный логин или пароль';
            return;

        saveSession(user);
        loginError.textContent = '';

        showToast(`Добро пожаловать, ${user.name || user.login}!`);
